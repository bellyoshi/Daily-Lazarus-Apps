1. 駒の移動定義（CanMoveロジック）
まず、各駒が「その相対座標に移動できるか」を判定する関数を作ります。量子将棋では、成っているかどうかで動きが劇的に変わるため、IsPromoted のフラグが重要です。

Delphi
function CanMove(pt: TPieceType; dx, dy: Integer; IsPromoted: Boolean; Owner: Integer): Boolean;
var
  absX, absY: Integer;
begin
  // 先手(0)は上(dy < 0)へ、後手(1)は下(dy > 0)へ進む。
  // 計算を単純にするため、後手の場合は座標を反転させる。
  if Owner = 1 then
  begin
    dy := -dy;
    dx := -dx;
  end;

  absX := Abs(dx);
  absY := Abs(dy);

  // 成っている場合は、金または龍・馬の動きになる
  if IsPromoted then
  begin
    case pt of
      ptPawn, ptLance, ptKnight, ptSilver, ptGold: 
        // 成銀、成桂、成香、と金、および金はすべて「金」の動き
        Exit((absX <= 1) and (dy >= -1) and (dy <= 1) and not ((dy = 1) and (absX = 1)));
      ptBishop: // 馬（角+王）
        Exit(((absX = absY)) or ((absX <= 1) and (absY <= 1)));
      ptRook: // 龍（飛+王）
        Exit(((dx = 0) or (dy = 0)) or ((absX <= 1) and (absY <= 1)));
    end;
  end;

  // 通常時の動き
  case pt of
    ptPawn:   Exit((dx = 0) and (dy = -1));
    ptLance:  Exit((dx = 0) and (dy < 0));
    ptKnight: Exit((absX = 1) and (dy = -2));
    ptSilver: Exit(((absX = 1) and (dy = -1)) or ((dy = 1) and (absX = 1)) or ((dx = 0) and (dy = -1)));
    ptGold:   Exit((absX <= 1) and (dy >= -1) and (dy <= 1) and not ((dy = 1) and (absX = 1)));
    ptBishop: Exit(absX = absY);
    ptRook:   Exit((dx = 0) or (dy = 0));
    ptKing:   Exit((absX <= 1) and (absY <= 1) and not ((dx = 0) and (dy = 0)));
    else Result := False;
  end;
end;

  pt: TPieceType;

駒台にある量子状態の駒を、空いているマスに打ちます。この際、「打つ場所」によって可能性がさらに絞り込まれます（例：行き所のない駒、二歩の禁止）。

Delphi
procedure TGameController.DropPiece(PieceIndex: Integer; toX, toY: Integer);
var
  Piece: TQuantumPiece;
  NewPossibilities: TPieceSet;
begin
  Piece := Stands[CurrentPlayer].Items[PieceIndex];

  // 1. 打つ場所によるフィルタリング（観測）
  NewPossibilities := Piece.Possibilities;

  // ルール：行き所のない駒（一番奥に歩や香を打ってはいけない）
  if (CurrentPlayer = 0) and (toY = 1) then
    NewPossibilities := NewPossibilities - [ptPawn, ptLance];
  if (CurrentPlayer = 0) and (toY <= 2) then
    NewPossibilities := NewPossibilities - [ptKnight];

  // ルール：二歩の禁止
  // その列に、既に「歩」として確定している自分の駒があれば、この駒は歩になれない
  if HasConfirmedPawnInColumn(toX, CurrentPlayer) then
    NewPossibilities := NewPossibilities - [ptPawn];

  // 2. 矛盾チェック
  if NewPossibilities = [] then
  begin
    window.alert('矛盾が発生するため、そこには打てません');
    Exit;
  end;

  // 3. 状態更新と配置
  Piece.Possibilities := NewPossibilities;
  Board[toX, toY] := Piece;
  Stands[CurrentPlayer].Delete(PieceIndex); // 駒台から消す

  UIBoard.Refresh(Self);
end;

通常の将棋では「王」は最初から1枚ですが、量子将棋では**「どの駒も王である可能性がある」**という前提で、移動の矛盾から王の正体を追い詰めていきます。

1. 「王（玉）」の特殊な性質
プログラム上、王には以下の制約を課します。

唯一性: 盤上と駒台を通じて、自陣営に「王」である可能性を持つ駒は常に最低1つ存在しなければならない。

非捕獲性: 相手が次に「王である可能性がある駒」を確実に取れる状態（王手）を放置してはならない。

2. 王の存在確率（存在可能性）の管理
各駒の Possibilities から ptKing が消えるタイミングを厳密に計算します。

Delphi
procedure TGameController.CheckKingFeasibility;
var
  x, y, KingPossibleCount: Integer;
begin
  KingPossibleCount := 0;
  
  // 盤上の自分の駒をすべてチェック
  for y := 1 to 9 do
    for x := 1 to 9 do
      if (Board[x, y] <> nil) and (Board[x, y].Owner = CurrentPlayer) then
        if ptKing in Board[x, y].Possibilities then
          Inc(KingPossibleCount);
          
  // 駒台の自分の駒もチェック
  for Piece in Stands[CurrentPlayer] do
    if ptKing in Piece.Possibilities then
      Inc(KingPossibleCount);

  // もし自分の「王」の可能性が0になったら、その瞬間に負け（矛盾負け）
  if KingPossibleCount = 0 then
    EndGame(Winner := 1 - CurrentPlayer, '自軍に王が存在しなくなりました');
end;
3. 王手（Check）の判定
量子将棋の王手は「確率的」です。**「相手の駒を次の手で取れる可能性があり、その駒が王である可能性がある」**場合に発生します。

Delphi
function TGameController.IsKingInDanger(Player: Integer): Boolean;
var
  tx, ty, fx, fy: Integer;
  Attacker: TQuantumPiece;
begin
  Result := False;
  // 相手(1-Player)の全駒の動きをシミュレーション
  for fy := 1 to 9 do
    for fx := 1 to 9 do
    begin
      Attacker := Board[fx, fy];
      if (Attacker <> nil) and (Attacker.Owner <> Player) then
      begin
        // 攻撃側の駒が「王を取れる動き」を持っているか
        for ty := 1 to 9 do
          for tx := 1 to 9 do
          begin
            if (Board[tx, ty] <> nil) and (Board[tx, ty].Owner = Player) and
               (ptKing in Board[tx, ty].Possibilities) then
            begin
              // その攻撃駒が(fx,fy)から(tx,ty)へ動ける可能性があるか
              if CalculateNewPossibilities(Attacker, tx-fx, ty-fy) <> [] then
                Exit(True); // 王を狙える駒が存在する
            end;
          end;
      end;
    end;
end;
4. 観測による勝敗の確定
最もエキサイティングなのは、駒を**「取った」**瞬間です。

王の可能性がある駒を取る:

その駒に「王以外」の可能性（歩や飛など）が残っていれば、その駒は「王ではなかった」とみなされ、ゲームは続行します（Possibilities から ptKing が除去される）。

もしその駒が**「王である可能性しかなかった」、あるいは「他の全駒が王ではないと確定していた」**場合、その瞬間にゲームセットです。

procedure TGameController.FinalizeTurn;

  CheckKingFeasibility();

  end;
  if CurrentPlayer = 0 then document.getElementById('status').innerText := '手番: 先手'
  else document.getElementById('status').innerText := '手番: 後手';
end;

procedure TGame.MovePiece(fx, fy, tx, ty: Integer);
var
  p: TQuantumPiece;
  newSet: TPieceSet;
  pt: TPieceType;
begin
  p := Board[fx, fy];
  newSet := [];
  for pt in p.Possibilities do if CanMove(pt, tx-fx, ty-fy, p.Owner) then Include(newSet, pt);
  if newSet = [] then begin window.alert('矛盾移動です'); Exit; end;
  p.Possibilities := newSet;
  if Board[tx, ty] <> nil then begin
    Board[tx, ty].Owner := p.Owner;
    Stands[p.Owner].Add(Board[tx, ty]);
  end;
  Board[tx, ty] := p; Board[fx, fy] := nil;
  CheckVictory;
  CurrentPlayer := 1 - CurrentPlayer;
  UpdateUI;
end;

procedure TGame.HandleCellClick(ax, ay: Integer);
begin
  if GameOver then Exit;
  if SelectedX = -1 then begin
    if (Board[ax, ay] <> nil) and (Board[ax, ay].Owner = CurrentPlayer) then begin
      SelectedX := ax; SelectedY := ay; UpdateUI;
    end;
  end else begin
    if (SelectedX = ax) and (SelectedY = ay) then begin SelectedX := -1; SelectedY := -1; UpdateUI; end
    else begin MovePiece(SelectedX, SelectedY, ax, ay); SelectedX := -1; SelectedY := -1; end;
  end;
end;

function OnCellClick(e: TJSEvent): boolean;
var
  t: TJSHTMLElement;
begin
  t := TJSHTMLElement(e.currentTarget);
  Game.HandleCellClick(StrToInt(t.dataset['x']), StrToInt(t.dataset['y']));
  Result := true;
end;

TJSHTMLElement(e.currentTarget): TJSEvent 型で受け取った後、中身を TJSHTMLElement にキャストして dataset（座標情報）にアクセスしています。
